% --- Окружение для переносимого листинга с кодом внутри документа ---
\newtcblisting[auto counter, number within=chapter]{breakablecode}[2][]{
	listing only,
	breakable,
	enhanced,
	sharp corners,
	colback=white,
	colframe=black,
	boxrule=0.5pt,
	left=10pt,
	right=5pt,
	top=10pt,
	bottom=5pt,
	before skip=10pt,
	after skip=10pt,
	%title={Листинг~\thetcbcounter~---~#2},
	listing style=myCodeStyle,
	#1,
	% --- Обеспечиваем рисование полных рамок при разрыве ---
	overlay first={
		\draw[line width=.5pt] (frame.south west)--(frame.south east);
	},
	overlay middle={
		\draw[line width=.5pt] (frame.south west)--(frame.south east);
		\draw[line width=.5pt] (frame.north west)--(frame.north east);
	},
	overlay last={
		\draw[line width=.5pt] (frame.north west)--(frame.north east);
	},
	% --- Если нужно на страницах-продолжениях слово "Продолжение" ---
	% Раскомментируйте строки ниже, если хотите, чтобы на второй и последующих страницах
	% вместо полного заголовка было только "Продолжение листинга N".
	% В этом случае закомментируйте опцию title выше.
	title={},
	overlay first={
		     \node[anchor=north west, xshift=5pt, yshift=-5pt] at (frame.north west) 
		         {Листинг~\thetcbcounter: #2};
		 },
	 overlay middle={
	     \node[anchor=north west, xshift=5pt, yshift=-5pt] at (frame.north west) 
	         {Продолжение листинга~\thetcbcounter~---~#2};
	 }
	overlay last={
	    \node[anchor=north west, xshift=5pt, yshift=-5pt] at (frame.north west) 
	        {Продолжение листинга~\thetcbcounter~---~#2};
	}
}
